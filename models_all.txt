class User(AbstractUser):
    """Customers, staff, and admins are all Users.

    Role is determined by is_superuser and StaffAssignment rows, never by a
    free-form field a client could tamper with.
    """

    email = models.EmailField("email address", unique=True)
    phone = models.CharField(max_length=20, blank=True)

    def save(self, *args, **kwargs):
        # Normalise so "A@x.com" and "a@x.com" can't be two accounts.
        self.email = self.email.strip().lower()
        super().save(*args, **kwargs)

    def __str__(self):
        return self.get_full_name() or self.username

def validate_timezone(value):
    try:
        ZoneInfo(value)
    except (ZoneInfoNotFoundError, ValueError):
        raise ValidationError(f"'{value}' is not a valid IANA time zone (e.g. Asia/Kolkata).")


def _random_name(folder, filename):
    ext = Path(filename).suffix.lower()
    return f"{folder}/{uuid.uuid4().hex}{ext}"


def property_image_path(instance, filename):
    return _random_name("properties", filename)



def room_image_path(instance, filename):
    return _random_name("rooms", filename)

class Property(TimeStampedModel):
    name = models.CharField(max_length=200)
    slug = models.SlugField(max_length=220, unique=True)

    address_line = models.CharField(max_length=255)

    email = models.EmailField()

    # Decision 5: one time zone per property.
    timezone = models.CharField(max_length=64, default="Asia/Kolkata", validators=[validate_timezone])
    # Decision 6: per-property standard times.
    check_in_time = models.TimeField(default="14:00")

    # Decision 3: cancellation window. Decision 9: one tax rate.
    free_cancellation_hours = models.PositiveIntegerField(default=24)

        max_digits=5, decimal_places=2, default=0,
        validators=[MinValueValidator(0), MaxValueValidator(100)],

    is_active = models.BooleanField(default=True, db_index=True)

    class Meta:
        verbose_name_plural = "properties"
        ordering = ["name"]
        constraints = [
            models.CheckConstraint(
                condition=Q(tax_percent__gte=0) & Q(tax_percent__lte=100),
                name="property_tax_percent_0_100",
            ),
        ]

    def __str__(self):
        return f"{self.name} ({self.city})"

    def clean(self):
        super().clean()
        # Amenities are many-to-many, so scope is checked in forms/services (Module 3).
    
    def local_today(self):
        return timezone.now().astimezone(ZoneInfo(self.timezone)).date()

class Room(TimeStampedModel):
    class OperationalStatus(models.TextChoices):
        ACTIVE = "ACTIVE", "Active"
        OUT_OF_SERVICE = "OUT_OF_SERVICE", "Out of service"  # long-term; short blocks use RoomBlock

    class HousekeepingStatus(models.TextChoices):
        CLEAN = "CLEAN", "Clean"
        DIRTY = "DIRTY", "Dirty"
        INSPECTING = "INSPECTING", "Inspecting"

    class SizeUnit(models.TextChoices):
        SQFT = "SQFT", "sq ft"
        SQM = "SQM", "sq m"

    property = models.ForeignKey(Property, on_delete=models.PROTECT, related_name="rooms")
    room_type = models.ForeignKey(RoomType, on_delete=models.PROTECT, related_name="rooms")
    number = models.CharField(max_length=20)
    floor = models.SmallIntegerField(null=True, blank=True)
    description = models.TextField(blank=True)

    max_guests = models.PositiveSmallIntegerField(default=2)
    base_occupancy = models.PositiveSmallIntegerField(default=2)
    bed_count = models.PositiveSmallIntegerField(default=1)
    bed_type = models.ForeignKey(BedType, on_delete=models.PROTECT, related_name="rooms")
    size = models.DecimalField(max_digits=7, decimal_places=2, null=True, blank=True)
    size_unit = models.CharField(max_length=4, choices=SizeUnit.choices, default=SizeUnit.SQFT)

    price_per_night = models.DecimalField(max_digits=10, decimal_places=2, validators=[MinValueValidator(0)])
    extra_guest_charge = models.DecimalField(
        max_digits=10, decimal_places=2, default=0, validators=[MinValueValidator(0)]
    )
    monthly_only = models.BooleanField(default=False, help_text="Bookable in whole months only")

    amenities = models.ManyToManyField(Amenity, blank=True, related_name="rooms")

    operational_status = models.CharField(
        max_length=20, choices=OperationalStatus.choices, default=OperationalStatus.ACTIVE, db_index=True
    )
    housekeeping_status = models.CharField(
        max_length=20, choices=HousekeepingStatus.choices, default=HousekeepingStatus.CLEAN
    )

    class Meta:
        ordering = ["property", "number"]
        constraints = [
            models.UniqueConstraint(fields=["property", "number"], name="uniq_room_number_per_property"),
            models.CheckConstraint(condition=Q(price_per_night__gte=0), name="room_price_non_negative"),
            models.CheckConstraint(condition=Q(extra_guest_charge__gte=0), name="room_extra_charge_non_negative"),
            models.CheckConstraint(condition=Q(max_guests__gte=1), name="room_max_guests_min_1"),
            models.CheckConstraint(condition=Q(base_occupancy__gte=1), name="room_base_occupancy_min_1"),
            models.CheckConstraint(
                condition=Q(base_occupancy__lte=F("max_guests")), name="room_base_occupancy_lte_max"
            ),
        ]
        indexes = [models.Index(fields=["property", "operational_status"])]

    def __str__(self):
        return f"{self.property.name} - Room {self.number}"

REF_ALPHABET = "ABCDEFGHJKMNPQRSTUVWXYZ23456789"   # no 0/O/1/I/L


def generate_reference():
    return "HB-" + "".join(secrets.choice(REF_ALPHABET) for _ in range(8))


class BookingQuerySet(models.QuerySet):
    def overlapping(self, check_in, check_out):
        return self.filter(check_in__lt=check_out, check_out__gt=check_in)

    def blocking(self, now=None):
        now = now or timezone.now()
        always = [Booking.Status.CONFIRMED, Booking.Status.CHECKED_IN, Booking.Status.CHECKED_OUT]
        return self.filter(
            Q(status__in=always)
            | Q(status=Booking.Status.PENDING, expires_at__isnull=True)
            | Q(status=Booking.Status.PENDING, expires_at__gt=now)
        )


class Booking(TimeStampedModel):
    class Status(models.TextChoices):
        PENDING = "PENDING", "Pending"
        CONFIRMED = "CONFIRMED", "Confirmed"
        CHECKED_IN = "CHECKED_IN", "Checked in"
        CHECKED_OUT = "CHECKED_OUT", "Checked out"
        CANCELLED = "CANCELLED", "Cancelled"
        NO_SHOW = "NO_SHOW", "No show"
        EXPIRED = "EXPIRED", "Expired"

    class PaymentStatus(models.TextChoices):
        UNPAID = "UNPAID", "Unpaid"
        PENDING = "PENDING", "Pending"
        PARTIALLY_PAID = "PARTIALLY_PAID", "Partially paid"
        PAID = "PAID", "Paid"
        REFUNDED = "REFUNDED", "Refunded"
        FAILED = "FAILED", "Failed"
        EXTENSION = "EXTENSION", "Extension"
        ADJUSTMENT = "ADJUSTMENT", "Adjustment"

    class Source(models.TextChoices):
        ONLINE = "ONLINE", "Online"
        ADMIN_MANUAL = "ADMIN_MANUAL", "Admin manual"
        WALK_IN = "WALK_IN", "Walk-in"

    reference = models.CharField(max_length=20, unique=True, default=generate_reference, editable=False)
    # One payment can cover several rooms; Module 5 fills this in.
    group_ref = models.CharField(max_length=20, blank=True, db_index=True)

    property = models.ForeignKey(Property, on_delete=models.PROTECT, related_name="bookings")
    room = models.ForeignKey(Room, on_delete=models.PROTECT, related_name="bookings")
    customer = models.ForeignKey(
        settings.AUTH_USER_MODEL, null=True, blank=True, on_delete=models.SET_NULL, related_name="bookings"
    )
    customer_name = models.CharField(max_length=200)
    customer_email = models.EmailField()
    customer_phone = models.CharField(max_length=20)

    check_in = models.DateField()
    check_out = models.DateField()
    guests = models.PositiveSmallIntegerField(default=1)

    # Snapshots taken at booking time; later rate changes never rewrite history.
    price_per_night = models.DecimalField(max_digits=10, decimal_places=2)
    additional_charges = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    discount = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    tax_amount = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    total_amount = models.DecimalField(max_digits=12, decimal_places=2)

    status = models.CharField(max_length=15, choices=Status.choices, default=Status.PENDING, db_index=True)
    payment_status = models.CharField(max_length=15, choices=PaymentStatus.choices, default=PaymentStatus.UNPAID)
    source = models.CharField(max_length=15, choices=Source.choices, default=Source.ONLINE)
    expires_at = models.DateTimeField(null=True, blank=True)   # hold deadline for PENDING
    amount_paid = models.DecimalField(max_digits=12, decimal_places=2, default=0,
                                      help_text="Only meaningful while PARTIALLY_PAID")
    actual_check_in = models.DateTimeField(null=True, blank=True)
    actual_check_out = models.DateTimeField(null=True, blank=True)
    checked_in_by = models.ForeignKey(settings.AUTH_USER_MODEL, null=True, blank=True,
                                      on_delete=models.SET_NULL, related_name="+")
    checked_out_by = models.ForeignKey(settings.AUTH_USER_MODEL, null=True, blank=True,
                                       on_delete=models.SET_NULL, related_name="+")
    special_requests = models.TextField(blank=True)
    internal_notes = models.TextField(blank=True)
    created_by = models.ForeignKey(
        settings.AUTH_USER_MODEL, null=True, blank=True, on_delete=models.SET_NULL, related_name="+"
    )

    objects = BookingQuerySet.as_manager()

    class Meta:
        ordering = ["-created_at"]
        constraints = [
            models.CheckConstraint(condition=Q(check_out__gt=F("check_in")), name="booking_checkout_after_checkin"),
            models.CheckConstraint(condition=Q(guests__gte=1), name="booking_guests_min_1"),
            models.CheckConstraint(
                condition=Q(price_per_night__gte=0) & Q(total_amount__gte=0)
                & Q(additional_charges__gte=0) & Q(discount__gte=0) & Q(tax_amount__gte=0),
                name="booking_amounts_non_negative",
            ),
            models.CheckConstraint(condition=Q(amount_paid__gte=0), name="booking_amount_paid_non_negative"),
        ]
        indexes = [
            models.Index(fields=["room", "check_in", "check_out"]),   # overlap lookups
            models.Index(fields=["status", "expires_at"]),            # hold-expiry job
        ]

    @cached_property
    def nights(self):
        # Derived, not stored, so it can never disagree with the dates.
        return (self.check_out - self.check_in).days

    def __str__(self):
        return f"{self.reference} ({self.room})"


class RoomBlock(TimeStampedModel):

    room = models.ForeignKey(Room, on_delete=models.CASCADE, related_name="blocks")
    start_date = models.DateField()
    end_date = models.DateField()
    reason = models.CharField(max_length=255)
    created_by = models.ForeignKey(
        settings.AUTH_USER_MODEL, null=True, blank=True, on_delete=models.SET_NULL, related_name="+"
    )

    class Meta:
        ordering = ["start_date"]
        constraints = [
            models.CheckConstraint(condition=Q(end_date__gt=F("start_date")), name="block_end_after_start"),
        ]
        indexes = [models.Index(fields=["room", "start_date", "end_date"])]

    def __str__(self):
        return f"{self.room} blocked {self.start_date} to {self.end_date}"

                              on_delete=models.SET_NULL, related_name="+")

    details = models.JSONField(default=dict, blank=True)

    class Meta:

        indexes = [models.Index(fields=["booking", "created_at"])]

    def save(self, *args, **kwargs):
